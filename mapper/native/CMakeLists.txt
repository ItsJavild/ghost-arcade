cmake_minimum_required(VERSION 3.20)
project(mapper_ndi_receiver LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The receiver targets Windows x64 only")
endif()
find_path(NDI_INCLUDE Processing.NDI.Lib.h
  HINTS "$ENV{NDI_SDK_DIR}/Include" "$ENV{NDI_SDK_DIR}/include"
  "C:/Program Files/NDI/NDI SDK/Include" "C:/Program Files/NewTek/NDI SDK/Include")
if(NOT NDI_INCLUDE)
  message(FATAL_ERROR "Install the official standard NDI SDK; set NDI_SDK_DIR to its root")
endif()
execute_process(COMMAND node -p "require('node-addon-api').include" WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/.." OUTPUT_VARIABLE NAPI_INCLUDE OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
string(REPLACE "\"" "" NAPI_INCLUDE "${NAPI_INCLUDE}")
add_library(ndi_receiver SHARED receiver.cpp ${CMAKE_JS_SRC})
target_include_directories(ndi_receiver PRIVATE ${CMAKE_JS_INC} ${NAPI_INCLUDE} ${NDI_INCLUDE})
target_link_libraries(ndi_receiver PRIVATE ${CMAKE_JS_LIB})
target_compile_features(ndi_receiver PRIVATE cxx_std_17)
target_compile_definitions(ndi_receiver PRIVATE NAPI_VERSION=9 NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(ndi_receiver PROPERTIES PREFIX "" SUFFIX ".node")
if(MSVC)
  set_property(TARGET ndi_receiver PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
